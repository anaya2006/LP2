public class StudentController {

    // Object for form data
    public Student__c student { get; set; }

    // List to display records
    public List<Student__c> studentList { get; set; }

    // Check edit mode
    public Boolean isEdit { get; set; }

    // Constructor
    public StudentController() {

        student = new Student__c();

        isEdit = false;

        loadStudents();
    }

    // Load all students
    public void loadStudents() {

        studentList = [
            SELECT Id,
                   Name,
                   Age__c,
                   Email__c,
                   Course__c
            FROM Student__c
        ];
    }

    // Save or Update Student
    public void saveStudent() {

        if(isEdit) {

            update student;

        } else {

            insert student;
        }

        // Reset form
        student = new Student__c();

        isEdit = false;

        loadStudents();
    }

    // Edit Student
    public void editStudent() {

        String studentId =
            ApexPages.currentPage()
            .getParameters()
            .get('id');

        student = [
            SELECT Id,
                   Name,
                   Age__c,
                   Email__c,
                   Course__c
            FROM Student__c
            WHERE Id = :studentId
            LIMIT 1
        ];

        isEdit = true;
    }

    // Delete Student
    public void deleteStudent() {

        String studentId =
            ApexPages.currentPage()
            .getParameters()
            .get('id');

        Student__c s = [
            SELECT Id
            FROM Student__c
            WHERE Id = :studentId
            LIMIT 1
        ];

        delete s;

        loadStudents();
    }
}
